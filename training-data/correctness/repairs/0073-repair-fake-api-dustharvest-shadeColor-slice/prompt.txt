The following JavaScript throws "TypeError: ....sub is not a function" when run. Find the bug and return the complete corrected, runnable code.

```js
function shadeColor(hex, amt) {
  if (!hex || hex[0]!=='#') return hex;
  let r=parseInt(hex.sub(1,3),16), g=parseInt(hex.slice(3,5),16), b=parseInt(hex.slice(5,7),16);
  r=Math.max(0,Math.min(255,r+amt)); g=Math.max(0,Math.min(255,g+amt)); b=Math.max(0,Math.min(255,b+amt));
  return '#'+r.toString(16).padStart(2,'0')+g.toString(16).padStart(2,'0')+b.toString(16).padStart(2,'0');
}
```
