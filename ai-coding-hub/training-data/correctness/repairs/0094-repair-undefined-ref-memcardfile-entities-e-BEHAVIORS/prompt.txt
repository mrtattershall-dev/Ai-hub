```js
/**
 * enemy.js — data-driven enemies with simple, readable AI.
 *
 * One class covers every archetype; behaviour is selected by the `behavior`
 * field on the enemy definition (data/enemies.json) and expressed as an FSM:
 *
 *   chaser  idle -> chase -> windup -> attack -> (recover) ...   (Static Grunt)
 *   turret  idle -> aim -> fire ...                              (Caster Turret)
 *   brute   idle -> chase -> windup -> attack ; blocks frontal   (Shield Brute)
 *           light attacks, weak to slams and back-hits
 *
 * Enemies share the platformer physics so they stand on terrain, telegraph
 * attacks with a color flash (GDD: "enemies telegraph clearly"), and award
 * shards + style on death through `world.onEnemyKilled`.
 *
 * @module entities/enemy
 */

import { StateMachine } from "../engine/fsm.js";
import { moveAndCollide } from "../systems/physics.js";
import { PHYSICS } from "../config.js";
import { approach, clamp, aabb, rectCenter } from "../engine/mathx.js";

export class Enemy {
  /**
   * @param {import('../main.js').App} app
   * @param {object} def stats from data/enemies.json
   * @param {number} x @param {number} y
   */
  constructor(app, def, x, y) {
    this.app = app;
    this.def = def;
    this.type = def.archetype;
    this.w = def.w;
    this.h = def.h;
    this.x = x - def.w / 2;
    this.y = y - def.h;
    this.vx = 0;
    this.vy = 0;
    this.facing = -1;
    this.hp = def.hp;
    this.maxHp = def.hp;
    this.alive = true;
    this.grounded = false;
    this.iframes = 0;
    this.contactTimer = 0;
    this.attackTimer = 0;     // cooldown between attacks / shots
    this.flash = 0;           // white hit-flash timer
    this.telegraph = 0;       // 0..1 windup progress for rendering
    this.fsm = new StateMachine(this, BEHAVIORS[def.behavior] ?? BEHAVIORS.chaser, "idle");
  }

  get aabb() {
    return { x: this.x, y: this.y, w: this.w, h: this.h };
  }

  /**
   * Incoming damage from a player attack.
   * @returns {boolean} false if the hit was blocked (no damage dealt)
   */
  hurt(amount, knockX = 0, _launch = 0, type = "light") {
    if (!this.alive) return false;

    // Shield Brute: blocks light/air attacks coming from the side it faces.
    if (this.def.shielded && (type === "light" || type === "air")) {
      const attackerSide = -Math.sign(knockX); // where the attacker stands
      if (attackerSide === this.facing) {
        this.app.audio.sfx("cancel");
        this.app.bus.emit("enemy:blocked", this);
        return false;
      }
    }

    this.hp -= amount;
    this.flash = 0.12;
    this.vx = knockX;
    if (_launch) this.vy = _launch;
    this.iframes = 0.08;
    if (this.hp <= 0) {
      this.die();
    } else {
      this.fsm.change("hurt");
    }
    return true;
  }

  die() {
    this.alive = false;
    this.app.audio.sfx("hit");
    this.world?.onEnemyKilled?.(this);
    this.app.bus.emit("enemy:killed", this);
  }

  // ---- shared helpers -----------------------------------------------------

  _toPlayer(world) {
    const me = rectCenter(this.aabb);
    const pc = rectCenter(world.player.aabb);
    return { dx: pc.x - me.x, dy: pc.y - me.y, dist: Math.hypot(pc.x - me.x, pc.y - me.y) };
  }

  _gravity(dt) {
    this.vy = Math.min(this.vy + PHYSICS.gravity * dt, PHYSICS.maxFallSpeed);
  }

  /** Contact damage with a short cooldown so it doesn't drain HP every frame. */
  _touchPlayer(world, dt) {
    this.contactTimer = Math.max(0, this.contactTimer - dt);
    if (this.contactTimer > 0) return;
    if (aabb(this.aabb, world.player.aabb)) {
      const dir = Math.sign(world.player.x - this.x) || 1;
      if (world.player.takeDamage(this.def.contactDamage, dir * 260)) {
        this.contactTimer = 0.8;
      }
    }
  }

  /** Melee swing connects with the player if still in range during attack. */
  _meleeHit(world) {
    const reach = this.def.attackRange ?? 50;
    const box = {
      x: this.facing > 0 ? this.x + this.w - 8 : this.x - reach + 8,
      y: this.y + 6,
      w: reach,
      h: this.h - 6,
    };
    if (aabb(box, world.player.aabb)) {
      const dir = this.facing;
      world.player.takeDamage(this.def.attackDamage, dir * 320);
    }
  }

  update(dt, world) {
    this.world = world;
    if (!this.alive) return;
    this.flash = Math.max(0, this.flash - dt);
    this.iframes = Math.max(0, this.iframes - dt);
    this.attackTimer = Math.max(0, this.attackTimer - dt);

    this.fsm.update(dt);

    // All enemies obey gravity so they rest on terrain (turrets included);
    // stationary archetypes simply keep vx at 0.
    this._gravity(dt);
    moveAndCollide(this, world.solids, dt);

    this._touchPlayer(world, dt);
  }

  render(ctx, cam) {
    if (!this.alive) return;
    const x = Math.round(this.x - cam.x);
    const y = Math.round(this.y - cam.y);
    const d = this.def;

    // Telegraph: flash toward the windup color before attacking.
    let fill = d.color;
    if (this.flash > 0) fill = "#ffffff";
    else if (this.telegraph > 0) {
      fill = this.telegraph > 0.5 ? d.telegraph : d.color;
    }

    ctx.fillStyle = fill;
    ctx.fillRect(x, y, this.w, this.h);
    ctx.strokeStyle = d.edge;
    ctx.lineWidth = 2;
    ctx.strokeRect(x + 1, y + 1, this.w - 2, this.h - 2);

    // Eye / facing indicator.
    ctx.fillStyle = "#0a0710";
    const ex = this.facing > 0 ? x + this.w - 14 : x + 6;
    ctx.fillRect(ex, y + 12, 8, 8);

    // Shield plate for the brute.
    if (d.shielded) {
      ctx.fillStyle = "#cfe0ff";
      const sx = this.facing > 0 ? x + this.w - 4 : x - 4;
      ctx.fillRect(sx, y + 4, 8, this.h - 8);
    }

    // HP pip bar when damaged.
    if (this.hp < this.maxHp) {
      const w = this.w;
      ctx.fillStyle = "rgba(0,0,0,0.6)";
      ctx.fillRect(x, y - 8, w, 4);
      ctx.fillStyle = "#7dffb0";
      ctx.fillRect(x, y - 8, w * clamp(this.hp / this.maxHp, 0, 1), 4);
    }
  }
}

/* ------------------------------------------------------------------------- *
 *  Behaviour tables keyed by `def.behavior`.
 * ------------------------------------------------------------------------- */


// Brute = chaser behaviour (its slower speed/longer windup come from data).
BEHAVIORS.brute = BEHAVIORS.chaser;
```
