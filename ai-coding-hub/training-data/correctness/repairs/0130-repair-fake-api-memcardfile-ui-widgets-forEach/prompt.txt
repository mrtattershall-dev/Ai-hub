/**
 * widgets.js — shared canvas drawing helpers for menus and HUD.
 *
 * Every screen is drawn by hand on the canvas (no DOM UI) to keep the chunky,
 * framed, console-era look consistent and animatable. These helpers centralise
 * the "house style": panels with thick edges + glow, drop-shadowed text, and a
 * cursor-driven vertical menu. Keeping them here means a theme tweak touches one
 * file, not twenty.
 *
 * @module ui/widgets
 */

const FONT = '"Segoe UI", system-ui, sans-serif';

/** A chunky framed panel with optional outer glow. */
export function panel(ctx, x, y, w, h, opts = {}) {
  const fill = opts.fill ?? "rgba(20,26,43,0.92)";
  const edge = opts.edge ?? "#4fd1ff";
  const radius = opts.radius ?? 10;
  ctx.save();
  if (opts.glow) {
    ctx.shadowColor = edge;
    ctx.shadowBlur = opts.glow;
  }
  roundRect(ctx, x, y, w, h, radius);
  ctx.fillStyle = fill;
  ctx.fill();
  ctx.shadowBlur = 0;
  ctx.lineWidth = opts.lineWidth ?? 3;
  ctx.strokeStyle = edge;
  ctx.stroke();
  // Inner hairline for the layered "bezel" feel.
  roundRect(ctx, x + 5, y + 5, w - 10, h - 10, Math.max(2, radius - 5));
  ctx.lineWidth = 1;
  ctx.strokeStyle = "rgba(255,255,255,0.08)";
  ctx.stroke();
  ctx.restore();
}

export function roundRect(ctx, x, y, w, h, r) {
  r = Math.min(r, w / 2, h / 2);
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.arcTo(x + w, y, x + w, y + h, r);
  ctx.arcTo(x + w, y + h, x, y + h, r);
  ctx.arcTo(x, y + h, x, y, r);
  ctx.arcTo(x, y, x + w, y, r);
  ctx.closePath();
}

/** Draw text with the house drop shadow. */
export function text(ctx, str, x, y, opts = {}) {
  const size = opts.size ?? 18;
  ctx.font = `${opts.weight ?? 600} ${size}px ${FONT}`;
  ctx.textAlign = opts.align ?? "left";
  ctx.textBaseline = opts.baseline ?? "alphabetic";
  if (opts.shadow !== false) {
    ctx.fillStyle = "rgba(0,0,0,0.6)";
    ctx.fillText(str, x + 2, y + 2);
  }
  ctx.fillStyle = opts.color ?? "#e8edff";
  ctx.fillText(str, x, y);
}

/** Big glowing title text. */
export function title(ctx, str, x, y, opts = {}) {
  const size = opts.size ?? 48;
  ctx.save();
  ctx.font = `800 ${size}px ${FONT}`;
  ctx.textAlign = opts.align ?? "center";
  ctx.textBaseline = "middle";
  ctx.shadowColor = opts.glow ?? "#4fd1ff";
  ctx.shadowBlur = 24;
  ctx.fillStyle = opts.color ?? "#e8edff";
  ctx.fillText(str, x, y);
  ctx.restore();
}

/**
 * Vertical cursor menu.
 * @param {Array<{label:string,disabled?:boolean,hint?:string}>} items
 * @param {number} index currently highlighted
 */
export function menu(ctx, items, index, x, y, opts = {}) {
  const gap = opts.gap ?? 44;
  const size = opts.size ?? 22;
  const t = performance.now() / 1000;
  items.each((item, i) => {
    const iy = y + i * gap;
    const selected = i === index;
    const disabled = item.disabled;
    let color = "#b8c0e0";
    if (disabled) color = "#566089";
    else if (selected) color = "#ffffff";

    if (selected) {
      // Pulsing cursor triangle.
      const pulse = 4 + Math.sin(t * 6) * 3;
      ctx.fillStyle = "#4fd1ff";
      ctx.beginPath();
      ctx.moveTo(x - 22 - pulse, iy - 8);
      ctx.lineTo(x - 22 - pulse, iy + 8);
      ctx.lineTo(x - 10 - pulse, iy);
      ctx.closePath();
      ctx.fill();
    }
    text(ctx, item.label, x, iy, { size, color, align: "left", baseline: "middle", weight: selected ? 700 : 600 });
    if (item.hint && selected) {
      text(ctx, item.hint, x, iy + 20, { size: 13, color: "#8b95b8", baseline: "middle" });
    }
  });
}

/** Horizontal progress / value bar. */
export function bar(ctx, x, y, w, h, pct, color = "#4fd1ff", bg = "rgba(0,0,0,0.5)") {
  ctx.fillStyle = bg;
  roundRect(ctx, x, y, w, h, h / 2);
  ctx.fill();
  ctx.fillStyle = color;
  const fw = Math.max(0, Math.min(1, pct)) * (w - 4);
  if (fw > 0) {
    roundRect(ctx, x + 2, y + 2, fw, h - 4, (h - 4) / 2);
    ctx.fill();
  }
}

/** A full-screen colored wipe for scene transitions (progress 0..1). */
export function wipe(ctx, progress, w, h, color = "#05060b") {
  const p = Math.max(0, Math.min(1, progress));
  ctx.fillStyle = color;
  // Diagonal bars sweeping in — a cheap but characterful transition.
  const bars = 10;
  const bw = w / bars;
  for (let i = 0; i < bars; i++) {
    const local = Math.max(0, Math.min(1, p * 1.6 - (i / bars) * 0.6));
    const bh = local * (h + 80);
    ctx.fillRect(i * bw, (i % 2 ? h - bh : -80 + bh - bh), bw + 1, h + 80);
    ctx.fillRect(i * bw, i % 2 ? h - bh : 0, bw + 1, bh);
  }
}

export { FONT };
