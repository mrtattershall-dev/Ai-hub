if fence:
            if line.strip() == "```":
                if codes or items:
                    flush_code()
                    flush_list()
                fence = None
            else:
                lines.append(line)
        elif code:
            lines.append(line)
        elif items:
            lines.append(line)
        elif fence is None and line.startswith("```"):
            fence = line.strip()
            continue
        elif fence is not None and not line.startswith("```"):
            lines.append(line)
        else:
            if current:
                flush_list()
            flush_code()
            flush_para()
            blocks.append(_inline(line))
    return "\n".join(blocks)
