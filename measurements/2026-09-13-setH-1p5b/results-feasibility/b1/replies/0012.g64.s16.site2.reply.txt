def flush_ol():
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]

    def take_heading(m):
        level = len(m.group(1))
        body = m.group(2)
        blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
        return ""

    text = re.sub(r"^(#{1,3}) (.*)$", take_heading, text)

    def take_list_item(m):
        line = m.group(0).strip()
        if line.startswith("- "):
            items.append(line[2:].strip())
            return ""
        flush_list()
        ol_items.append(line.strip())
        return ""

    text = re.sub(r"^(?=- \d\.) (.*)$", take_list_item, text)

    def take_code(m):
        codes.append("<code>" + _escape(m.group(1)) + "</code>")
        return _CODE % (len(codes) - 1)

    text = re.sub(r"`([^`]*)`", take_code, text)

    def take_link(m):
        label = _escape(m.group(1))
        url = _escape(m.group(2)).replace('"', "&quot;")
        links.append('<a href="' + url + '">' + label + "</a>")
        return _LINK % (len(links) - 1)

    text = re.sub(r"\[([^\]]*)\]\(([^)]*)\)", take_link, text)
