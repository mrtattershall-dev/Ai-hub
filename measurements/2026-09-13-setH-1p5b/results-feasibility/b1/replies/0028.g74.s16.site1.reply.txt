fence = None

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]

    def flush_code():
        if codes:
            blocks.append("<pre><code>" + "\n".join(codes) + "</code></pre>")
            del codes[:]

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            continue
        if line.startswith("```"):
            if fence is not None:
                flush_code()
            fence = line[3:]
            continue
        if line.endswith("```"):
            if fence is not None:
                blocks.append(fence)
                flush_code()
                fence = None
            continue
        if fence:
            codes.append(line.strip())
            continue
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
