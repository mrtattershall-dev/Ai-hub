fence = None

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("```") and fence is None:
            fence = []
        elif line.strip() == "```" and fence:
            if len(fence) > 1:  # multi-line code block
                blocks.append("<pre><code>" + "\n".join(fence) + "</code></pre>")
            else:  # fenced code block on one line
                blocks.append("<pre><code>" + _inline(fence[0]) + "</code></pre>")
            fence = None
        elif fence is not None:
            fence.append(line.strip())
        else:
            flush_list()
            h = _is_heading(line)
            if h:
                flush_para()
                level, body = h
                blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
