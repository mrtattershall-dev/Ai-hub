if re.match(r"\d+\.\s", line):
            flush_para()
            flush_ordered_list()
            items.append(line[2:].strip())
            continue
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
