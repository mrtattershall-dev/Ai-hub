if items:
                flush_ordered_list()
            ordered_items.append(line[2:].strip())
        elif re.match(r"^[0-9]+\.\s", line):
            flush_para()
            if items:
                flush_ordered_list()
            ordered_items.append(line[3:].strip())
        else:
            flush_ordered_list()
            current.append(line.strip())

    flush_ordered_list()

    # END SOLUTION
    return "\n".join(blocks)
