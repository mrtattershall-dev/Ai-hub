# BEGIN SOLUTION
    fenced_code_block = None
    # END SOLUTION

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]

    for line in str(text).split("\n"):
        # BEGIN SOLUTION
        if fenced_code_block is None:
            if line.strip() == "```":
                fenced_code_block = []
            elif line.startswith("- "):
                flush_para()
                items.append(line[2:].strip())
                continue
            elif h := _is_heading(line):
                flush_para()
                level, body = h
                blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
                continue
            else:
                current.append(line.strip())
        else:
            if line.strip() == "```":
                blocks.append("<pre><code>" + "\n".join(fenced_code_block) + "</code></pre>")
                fenced_code_block = None
            else:
                fenced_code_block.append(line)
        # END SOLUTION
    flush_para()
    flush_list()
    return "\n".join(blocks)
