    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]
        # ADD THE FOLLOWING CODE BLOCK TO FLUSH ORDERED LISTS
        if ordered_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ordered_items) + "</ol>")
            del ordered_items[:]

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        # CHECK FOR ORDERED LIST ITEMS AND HANDLE THEM
        m = re.match(r"(\d+)\. (.*)", line)
        if m:

            ordered_items.append(m.group(2).strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
