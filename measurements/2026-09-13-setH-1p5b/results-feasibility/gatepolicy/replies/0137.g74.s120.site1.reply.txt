fenced_code_block = None

    def flush_code():
        if fenced_code_block:
            blocks.append("<pre><code>" + "\n".join(fenced_code_block) + "</code></pre>")
            del fenced_code_block[:]

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_code()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue

        if line.startswith("```"):
            if fenced_code_block is None:
                fenced_code_block = []
            else:
                flush_code()
                fenced_code_block = None
            continue

        if fenced_code_block is not None:
            fenced_code_block.append(line)
            continue

        current.append(line.strip())
    flush_para()
    flush_list()
    flush_code()
    return "\n".join(blocks)
