ol_items = []  # NEW: declare accumulator list for ordered-list items
    ol_count = None  # Track the current ordered list number

    def flush_ol():
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]

    for line in str(text).split("\n"):
        # Check for ordered list item
        m = re.match(r"(\d+)\. (.*)", line)
        if m:
            flush_para()  # Flush any existing paragraph
            flush_ol()  # Flush any existing unordered list

            num, body = int(m.group(1)), m.group(2).strip()
            if ol_count is None or ol_count != num:  # New ordered list starting
                ol_items.append(body)
                ol_count = num
            else:  # Continuing current ordered list
                ol_items[-1] += " " + body

            continue

        flush_ol()  # Flush any existing ordered list
        if line.strip() == "":
            flush_para()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ol()  # Flush any remaining ordered list
    return "\n".join(blocks)
