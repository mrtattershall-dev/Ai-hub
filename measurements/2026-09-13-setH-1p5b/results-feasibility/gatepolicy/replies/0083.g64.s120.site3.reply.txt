if items:  # If an unordered list is open
                flush_ordered_list()  # Close it
            ol_items.append(line.strip())  # Add the line to the ordered list accumulator
            continue

        if re.match(r"^\d+\. ", line):  # Check for ordered list item
            flush_para()
            ol_items.append(line[2:].strip())  # Add the line to the ordered list accumulator
            continue

        if line.startswith("1. "):
            flush_ordered_list()  # Close any open unordered list
            ol_items.append(line[3:].strip())  # Start a new ordered list with this item
            continue

        if items:  # If an unordered list is open
            flush_ordered_list()  # Close it
        ol_items = []  # Clear the accumulator for ordered lists

        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Ensure any remaining ordered list is closed

    return "\n".join(blocks)
