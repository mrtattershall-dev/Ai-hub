flush_ordered_list()
            continue
        if line.startswith("1. "):
            flush_para()
            items.append(line[3:].strip())
            continue
        flush_list()
        flush_ordered_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()
    return "\n".join(blocks)
