def flush_ol():
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]
    # END OF NEW FUNCTION

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ol()  # ADDED FOR ORDERED LISTS
            continue
        if re.match(r"^\d+\. ", line):  # ADDED FOR ORDERED LISTS
            flush_para()
            ol_items.append(line[2:].strip())
            continue
        flush_list()
        flush_ol()  # ADDED FOR ORDERED LISTS
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ol()  # ADDED FOR ORDERED LISTS
    return "\n".join(blocks)
