# END SOLUTION
        flush_ordered_list()
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_ordered_list()
        if line.strip().endswith("."):
            flush_para()
        elif line.strip() == "":
            flush_para()
            flush_list()
        else:
            current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)

import re

_CODE = "\x00CODE%d\x00"
_LINK = "\x00LINK%d\x00"


def _escape(s):
    return s.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")


def _inline(text):
    codes = []
    links = []

    def take_code(m):
        codes.append("<code>" + _escape(m.group(1)) + "</code>")
        return _CODE % (len(codes) - 1)

    text = re.sub(r"`([^`]*)`", take_code, text)

    def take_link(m):
        label = _escape(m.group(1))
        url = _escape(m.group(2)).replace('"', "&quot;")
        links.append('<a href="' + url + '">' + label + "</a>")
        return _LINK % (len(links) - 1)

    text = re.sub(r"\[([^\]]*)\]\(([^)]*)\)", take_link, text)

    text = _escape(text)
    text = re.sub(r"\*\*(.+?)\*\*", r"<strong>\1</
