ol_items = []  # Accumulator list for ordered-list items
    # END OF REQUESTED CHANGE

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]
        # Flush ordered list items if any
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]

    for line in str(text).split("\n"):
        if line.strip() == "":

            flush_list()

        if m:

            ol_items.append(m.group(2))

        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
