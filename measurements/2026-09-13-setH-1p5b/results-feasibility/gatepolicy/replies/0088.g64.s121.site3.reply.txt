if ol_items:
                flush_ol()
            blocks.append("<p>")
        elif line.startswith("1. "):
            flush_para()
            ol_items.append(line[3:].strip())
            continue
        elif line == "":
            flush_ol()  # NEW LINE ADDED FOR FLUSHING ORDERED-LIST ITEMS ON BLANK LINES
            if items:
                blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
                del items[:]
            continue
        elif line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ol()  # NEW LINE ADDED FOR FLUSHING ORDERED-LIST ITEMS AT END OF TEXT
    return "\n".join(blocks)
