def flush_ol():
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]

    # END OF NEW CODE
    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ol()  # NEW: Flush the ordered list when a blank line is encountered
            continue
        if re.match(r"^\d+\.\s+", line):  # NEW: Check for ordered list item
            flush_para()
            flush_list()
            ol_items.append(line[2:].strip())  # Add the item to the accumulator list
            ol_count = None  # Reset the current ordered list number
            continue
        if re.match(r"^\d+\.\s+(\d+)\.", line):  # NEW: Check for continuation of an ordered list
            flush_para()
            flush_list()
            match = re.match(r"^\d+\.\s+(\d+)\.(.*)", line)
            ol_items.append(match.group(2).strip())  # Add the item to the accumulator list
            ol_count = int(match.group(1))  # Update the current ordered list number
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ol()  # NEW: Flush the ordered list when reaching the end of the input
    return "\n".join(blocks)
