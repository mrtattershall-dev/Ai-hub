if line.startswith("```"):
            flush_code()
            if fenced_code_block:
                blocks.append("<pre><code>" + "\n".join(fenced_code_block) + "</code></pre>")
                del fenced_code_block[:]
            else:
                fenced_code_block = []
            continue
        if fenced_code_block is not None:
            fenced_code_block.append(line)
            continue

        flush_para()
        flush_list()
        if line.strip() == "":
            blocks.append("<p></p>")
            continue
        if line.startswith("- "):
            items.append(line[2:].strip())
            continue
        h = _is_heading(line)
        if h:
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
