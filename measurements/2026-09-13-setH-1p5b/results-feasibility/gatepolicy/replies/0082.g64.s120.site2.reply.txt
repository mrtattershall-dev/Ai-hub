def flush_ordered_list():  # New function for ordered lists
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]  # Clear the accumulator

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ordered_list()  # Call new function
            continue
        if line.startswith("1. "):  # Check for ordered list item (new condition)
            flush_para()  # Ensure any previous paragraph is closed
            ol_items.append(line[3:].strip())  # Add the content of the ordered list item to the accumulator
            continue
        flush_list()
        flush_ordered_list()  # Call new function again to ensure it's cleared after each block
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Call new function one last time to ensure it's cleared after the final block
    return "\n".join(blocks)
