def flush_ol_list():
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]
    
    # Now integrate the new flush function into the main loop.
    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ol_list()
            continue
        m = re.match(r"^\d+\. (.*)$", line)
        if m:
            flush_para()
            ol_items.append(m.group(1).strip())
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ol_list()
    return "\n".join(blocks)
