def flush_ordered_list():
        if ordered_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ordered_items) + "</ol>")
            del ordered_items[:]  # Clear the list after flushing

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ordered_list()  # Flush any remaining ordered list items
            continue
        if re.match(r"^\d+\. ", line):  # Check if the line starts with a number, a dot and a space
            flush_para()  # Ensure current paragraph is flushed
            ordered_items.append(line[2:].strip())  # Extract and add to ordered list items
            continue
        flush_list()
        flush_ordered_list()  # Flush any remaining ordered list items
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Flush any remaining ordered list items
    return "\n".join(blocks)
