fenced_code_block = None
    # END OF NEW CODE

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        # START OF NEW CODE
        if line.strip() == "```":
            if fenced_code_block is None:
                fenced_code_block = []
            else:
                blocks.append("<pre><code>\n" + "\n".join(fenced_code_block) + "\n</code></pre>")
                del fenced_code_block[:]
                continue
        elif fenced_code_block is not None:
            fenced_code_block.append(line.strip())
            continue
        # END OF NEW CODE
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
