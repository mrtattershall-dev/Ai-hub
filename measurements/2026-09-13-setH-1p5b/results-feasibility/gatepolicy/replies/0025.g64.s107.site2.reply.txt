# BEGIN SOLUTION
    def flush_ordered_list():
        if ordered_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ordered_items) + "</ol>")
            del ordered_items[:]
    # END SOLUTION

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ordered_list()  # Call the new function to handle ordered lists
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        if re.match(r"^\d+\.\s", line):  # Check for ordered list item
            flush_para()
            flush_list()  # Flush unordered list before adding to ordered list
            ordered_items.append(line[3:].strip())  # Add the item to the accumulator
            continue
        flush_list()
        flush_ordered_list()  # Call the new function to handle ordered lists
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Call the new function to handle ordered lists
    return "\n".join(blocks)
