ol_items = []  # Accumulator for ordered-list items
    # END REQUESTED CHANGE
    flush_para = lambda: None
    flush_list = lambda: None

    def flush_ol():
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]  # Clear the accumulator for ordered-list items
            return True
        return False

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        # REQUESTED CHANGE: Add ordered lists to to_html in the EXISTING s4_markdown.py: consecutive lines starting with a number, a dot and a space ("1. ") become <ol><li>...</li>...</ol>, written like the unordered lists.
        elif line.strip().startswith("1. "):
            flush_para()
            ol_items.append(line[3:].strip())
            continue
        # END REQUESTED CHANGE
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    # Flush any remaining ordered-list items
    flush_ol()
    return "\n".join(blocks)
