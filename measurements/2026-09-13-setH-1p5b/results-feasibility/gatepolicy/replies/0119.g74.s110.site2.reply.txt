if fenced_code_block is None and line.startswith("```"):
            if current:
                flush_para()
            items = []
            continue
        elif fenced_code_block is not None:
            if line.strip() == "```":
                blocks.append("<pre><code>" + _escape("\n".join(items)) + "</code></pre>")
                items = []
                fenced_code_block = None
                continue
            else:
                items.append(line)
                continue
        elif line.startswith("```"):
            if current:
                flush_para()
            items = []
            fenced_code_block = True
            continue

        # END OF NEW CODE ADDED BY REQUEST
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
