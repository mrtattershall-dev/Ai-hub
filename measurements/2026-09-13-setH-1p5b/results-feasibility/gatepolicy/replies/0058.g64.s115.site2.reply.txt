def flush_ordered_list():
        if ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]

    # END SOLUTION

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ordered_list()  # Flush ordered list when a blank line is encountered
            continue
        m = re.match(r"(\d+)\. (.*)", line)
        if m:  # Check for ordered list item
            flush_para()  # Ensure no remaining paragraph from previous lines
            ol_items.append(m.group(2).strip())
            continue
        if line.startswith("- "):
            flush_ordered_list()  # Flush any open ordered list before starting a new one
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_ordered_list()  # Ensure no remaining ordered list from previous lines
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Flush any remaining ordered list at the end of the text
    return "\n".join(blocks)
