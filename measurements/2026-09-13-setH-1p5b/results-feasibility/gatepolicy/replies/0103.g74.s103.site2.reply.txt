if fenced_code_block is None:
            if line.strip() == "```":
                if current or items:
                    flush_para()
                    flush_list()
                fenced_code_block = []
                continue
        elif line.strip() != "```":
            fenced_code_block.append(line)
            continue
        else:
            blocks.append("<pre><code>" + _escape("\n".join(fenced_code_block)) + "</code></pre>")
            fenced_code_block = None
            continue
        # END OF ADDED CODE
        if line.strip() == "```":
            continue
        if current and current[-1] == "":
            del current[-1]
        flush_list()
        if fenced_code_block is not None:
            fenced_code_block.append(line)
            continue
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
