if items:
                flush_ordered_list()
            ol_items.append(line[2:].strip())
            continue
        if re.match(r"^\d+\. ", line):
            flush_para()
            # BEGIN SOLUTION
            if ol_items:
                flush_ordered_list()
            ol_items.append(line[3:].strip())
            # END SOLUTION
            continue
        flush_ordered_list()  # Added to close any open ordered list before starting a new paragraph or heading
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    # BEGIN SOLUTION
    flush_ordered_list()  # Added to close any open ordered list at the end of the document
    # END SOLUTION
    return "\n".join(blocks)
