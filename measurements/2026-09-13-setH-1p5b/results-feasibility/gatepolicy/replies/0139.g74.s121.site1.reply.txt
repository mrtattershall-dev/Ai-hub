fenced_code_block = None  # Declare the fenced-code-block state here
    for line in text.split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        # Process fenced code block
        if line.startswith("```"):
            if fenced_code_block is None:
                # Start a new fenced code block
                fenced_code_block = []
            else:
                # End the current fenced code block and flush it
                blocks.append("<pre><code>" + "\n".join(fenced_code_block) + "</code></pre>")
                fenced_code_block = None
        elif fenced_code_block is not None:
            # Collect lines for the current fenced code block
            fenced_code_block.append(line)
        else:
            current.append(line.strip())
    if fenced_code_block is not None:
        # Flush any remaining lines of an unclosed fenced code block
        blocks.append("<pre><code>" + "\n".join(fenced_code_block) + "</code></pre>")
    return "\n".join(blocks)
