flush_para()
        flush_ordered_list()  # Flush any existing ordered list before starting a new one
        items.append(line[2:].strip())  # Add the line to the unordered list
        continue

        m = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())

    flush_para()
    flush_list()
    flush_ordered_list()  # Flush any remaining ordered list at the end of the text

    return "\n".join(blocks)
