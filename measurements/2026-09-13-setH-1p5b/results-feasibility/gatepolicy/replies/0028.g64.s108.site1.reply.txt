ol_items = []  # Declare the accumulator list for ordered-list items
    # END REQUESTED CHANGE

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]
        # BEGIN REQUESTED CHANGE
        if ol_items:  # Check if there are any ordered-list items to flush
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]  # Clear the list after flushing
        # END REQUESTED CHANGE

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):  # Check for ordered lists starting with a number, dot and space
            match = re.match(r"(\d+)\. (.*)", line)
            if match:
                flush_para()
                ol_items.append(match.group(2).strip())
                continue
            flush_list()  # Fall back to unordered list if no ordered list pattern is found
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
