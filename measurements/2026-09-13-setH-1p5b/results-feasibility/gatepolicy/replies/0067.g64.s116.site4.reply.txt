if ordered_items:
                flush_ordered_list()
            ordered_items.append(line[2:].strip())
        elif re.match(r"^\d+\. ", line):  # Check if the line starts with a number, followed by a dot and a space
            flush_para()  # Flush any accumulated paragraph text
            if items:  # If there are unordered list items
                flush_list()  # Add them to blocks as an unordered list
            ordered_items.append(line[2:].strip())  # Add the line content (excluding the number, dot and space) to the ordered list accumulator
        else:
            flush_ordered_list()
            current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
