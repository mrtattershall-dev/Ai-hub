fenced_code_block = None
    # END OF NEW CODE
    for line in text.split("\n"):
        if line.strip() == "":
            flush_para()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        if fenced_code_block is not None:
            fenced_code_block.append(line)
            continue
        if line.startswith("```"):
            if line.endswith("```"):
                blocks.append("<pre><code>" + _escape("\n".join(fenced_code_block)) + "</code></pre>")
                fenced_code_block = None
            else:
                fenced_code_block = [line[3:]]
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(_escape(line.strip()))
    flush_para()
    flush_list()
    return "\n".join(blocks)
