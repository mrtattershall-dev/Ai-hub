# ADDITION STARTS HERE
    def flush_ordered_list():
        if ordered_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ordered_items) + "</ol>")
            del ordered_items[:]  # Clear the list after adding it to blocks

    # ADDITION ENDS HERE

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ordered_list()  # Flush ordered list when a blank line is encountered
            continue
        if re.match(r"^\d+\.\s", line):  # Check if the line starts with a number followed by a dot and a space
            flush_para()
            flush_list()
            items.append(line[3:].strip())

        h = _is_heading(line)
        if h:

            level, body = h

            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Flush any remaining ordered list items at the end of the text
    return "\n".join(blocks)
