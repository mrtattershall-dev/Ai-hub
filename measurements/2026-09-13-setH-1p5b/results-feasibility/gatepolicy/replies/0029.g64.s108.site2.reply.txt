# END OF REQUESTED CHANGE
    flush_ol = lambda: blocks.append("<ol>" + "".join("<li>" + _inline(i) for i in ol_items) + "</ol>") if ol_items else None

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ol()  # Flush any remaining ordered list items
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        if line.startswith("1. "):  # Check for ordered list item
            flush_para()
            ol_items.append(line[3:].strip())  # Add to the ordered list accumulator
            continue
        flush_list()
        flush_ol()  # Flush any remaining ordered list items before processing a new block
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ol()  # Flush any remaining ordered list items at the end of processing all lines
    return "\n".join(blocks)
