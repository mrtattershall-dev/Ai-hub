if line.strip().startswith("```"):
            flush_code()
            if fenced_code_block:
                blocks.append("<pre><code>" + "\n".join(fenced_code_block) + "</code></pre>")
                del fenced_code_block[:]
            else:
                fenced_code_block = []
            continue
        if fenced_code_block is not None:

        # REQUESTED CHANGE: Add fenced code blocks to to_html in the EXISTING s4_markdown.py: the lines between two lines of ``` become <pre><code>...</code></pre>, keeping their line breaks ("\n"), escaped, with no other formatting; a fence that is never closed runs to the end of the text.         # ALREADY HANDLED ELSEWHERE - do not add a second path for it: ordinary text is already accumulated by the existing current.append(line.strip()) at the end of the loop; paragraphs, headings and unordered lists already have their own branches
        # NOT AVAILABLE HERE (they are locals of other functions, not of to_html): codes, label, links, m, take_code, take_link, url
        # Any line this code does not handle must fall through to the EXISTING code unchanged.
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())

    flush_code()
    flush_para()
    flush_list()
    return "\n".join(blocks)
