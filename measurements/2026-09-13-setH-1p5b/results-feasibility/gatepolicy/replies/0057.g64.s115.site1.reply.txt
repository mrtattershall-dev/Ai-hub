# BEGIN SOLUTION
    ol_items = []
    # END SOLUTION

    def flush_para():
        if current:
            blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
            del current[:]

    def flush_list():
        if items:
            blocks.append("<ul>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ul>")
            del items[:]
        # BEGIN SOLUTION
        elif ol_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ol_items) + "</ol>")
            del ol_items[:]
        # END SOLUTION

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        # BEGIN SOLUTION
        m = re.match(r"(\d+)\. (.*)", line)
        if m:
            flush_para()
            ol_items.append(m.group(2).strip())
            continue
        # END SOLUTION
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
