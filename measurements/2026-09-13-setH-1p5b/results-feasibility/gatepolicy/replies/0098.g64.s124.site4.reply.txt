# If there are any open ordered list items, close them before starting a new unordered list.
            if ordered_items:
                flush_ordered_list()

         # END OF CHANGE

        if line.startswith("1. "):  # Check if the line is an ordered list item
            flush_para()  # Close any open paragraph before starting a new list
            flush_ordered_list()  # Close any open ordered list before starting a new one
            ordered_items.append(line[3:].strip())  # Add the item to the ordered list
            continue

        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
