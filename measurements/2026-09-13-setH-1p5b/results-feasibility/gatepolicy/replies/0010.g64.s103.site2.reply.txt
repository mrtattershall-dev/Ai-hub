def flush_ordered_list():
        if items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in items) + "</ol>")
            del items[:]
    # END REQUESTED CHANGE

    def take_code(m):
        codes.append("<code>" + _escape(m.group(1)) + "</code>")
        return _CODE % (len(codes) - 1)

    def take_link(m):
        label = _escape(m.group(1))
        url = _escape(m.group(2)).replace('"', "&quot;")
        links.append('<a href="' + url + '">' + label + "</a>")
        return _LINK % (len(links) - 1)

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        # REQUESTED CHANGE: Add ordered lists to to_html in the EXISTING s4_markdown.py: consecutive lines starting with a number, a dot and a space ("1. ") become <ol><li>...</li>...</ol>, written like the unordered lists. Run it with python.
        m = re.match(r"(\d+)\. (.*)", line)
        if m:
            flush_para()
            items.append(m.group(2).strip())
            continue
        # END REQUESTED CHANGE

        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())

    flush_para()
    flush_ordered_list()  # Flush ordered list after all other content processed
    flush_list()
    return "\n".join(blocks)
