flush_ol()
            blocks.append("<p>")
            continue
        if line.startswith("1. "):
            flush_para()
            ol_items.append(line[3:].strip())
            continue
        if line.startswith("# ") or line.startswith("## ") or line.startswith("### "):
            flush_para()
            flush_list()
            flush_ol()
            level = 1 + line.count('#')
            blocks.append("<h%d>%s</h%d>" % (level, _inline(line[level:].strip()), level))
            continue
        if line.startswith("**") and line.endswith("**"):
            flush_para()
            flush_list()
            flush_ol()
            blocks.append("<p><strong>%s</strong></p>" % _inline(line[2:-2]))
            continue
        if line.startswith("*") and line.endswith("*"):
            flush_para()
            flush_list()
            flush_ol()
            blocks.append("<p><em>%s</em></p>" % _inline(line[1:-1]))
            continue
        if line.startswith("`") and line.endswith("`"):
            flush_para()
            flush_list()
            flush_ol()
            blocks.append("<p><code>%s</code></p>" % _escape(line[1:-1]))
            continue
        if line.startswith("[") and line.endswith("]"):
            flush_para()
            flush_list()
            flush_ol()
            blocks.append('<p><a href="%s">%s</a></p>' % (re.match(r'\[(.+?)\]\((.+?)\)', line).group(2), re.match(r'\[(.+?)\]\((.+?)\)', line).group(1)))
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
