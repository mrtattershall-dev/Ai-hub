flush_ordered_list()
            current.append("")
        elif line.startswith("1. "):
            flush_para()
            ordered_items.append(line[3:].strip())
            continue
        if items:
            flush_ordered_list()
            continue
        if ordered_items:
            flush_ordered_list()
            continue
        if current and not current[-1]:
            blocks.append("<p>")
            del current[:]
        elif current:
            current.append("")
        flush_list()
        if line.startswith("# "):
            h = _is_heading(line)
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(_inline(line.strip()))

    return "\n".join(blocks)
