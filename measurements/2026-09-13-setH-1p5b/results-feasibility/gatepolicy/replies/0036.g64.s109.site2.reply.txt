def flush_ordered_list():  # ADD THIS FUNCTION
        if ordered_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ordered_items) + "</ol>")
            del ordered_items[:]  # Clear the list of ordered items after flushing

    for line in text.split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ordered_list()  # Flush any remaining ordered list items
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        if re.match(r"^\d+\. ", line):  # Check for ordered list item (e.g., "1. ")
            flush_para()
            ordered_items.append(line[3:].strip())  # Extract the content after number and dot
            continue
        flush_list()
        flush_ordered_list()  # Flush any remaining ordered list items before processing new block
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Flush any remaining ordered list items at the end of processing
    return "\n".join(blocks)
