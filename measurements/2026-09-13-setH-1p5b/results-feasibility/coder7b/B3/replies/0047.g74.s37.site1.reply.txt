fenced_code_block = None  # Declare the fenced-code-block state
    for line in text.split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        if line.startswith("```"):
            if fenced_code_block is None:
                # Start a new code block
                fenced_code_block = []
            else:
                # End the current code block and flush it
                blocks.append("<pre><code>{}</code></pre>".format("\n".join(fenced_code_block)))
                fenced_code_block = None
        elif fenced_code_block is not None:
            # Collect lines inside a code block
            fenced_code_block.append(line)
            continue
        current.append(line.strip())
    if fenced_code_block is not None:
        # Handle an unclosed code block at the end of the text
        blocks.append("<pre><code>{}</code></pre>".format("\n".join(fenced_code_block)))
    flush_para()
    flush_list()
    return "\n".join(blocks)
