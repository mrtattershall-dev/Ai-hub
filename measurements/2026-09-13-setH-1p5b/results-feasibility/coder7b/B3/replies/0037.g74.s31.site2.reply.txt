            fenced_code_block = []
        elif fenced_code_block is not None:
            if line.strip() == "```":
                blocks.append("<pre><code>" + "\n".join(_escape(l) for l in fenced_code_block) + "</code></pre>")
                fenced_code_block = None
            else:
                fenced_code_block.append(line)
        elif line.startswith("```"):
            flush_para()
            flush_list()
            lines = []
            while True:

                if line.strip() == "```":
                    break
                lines.append(line)
            blocks.append("<pre><code>" + "\n".join(_escape(l) for l in lines) + "</code></pre>")
        else:
            current.append(line.strip())
    flush_para()
    flush_list()
    return "\n".join(blocks)
