def flush_ordered_list():
        if ordered_items:
            blocks.append("<ol>" + "".join("<li>" + _inline(i) + "</li>" for i in ordered_items) + "</ol>")
            del ordered_items[:]  # Clear the list of ordered items

    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            flush_ordered_list()  # Add this to handle ordered lists
            continue
        m = re.match(r"^\d+\.\s(.*)", line)  # Check if the line starts with a number followed by ". "
        if m:
            flush_para()  # Flush any previous paragraph
            ordered_items.append(m.group(1).strip())  # Add the list item to the accumulator
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        flush_ordered_list()  # Ensure ordered lists are flushed before moving on to new content
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Flush any remaining ordered list items at the end of the text
    return "\n".join(blocks)
