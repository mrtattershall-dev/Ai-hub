if ordered_items:
                flush_ordered_list()  # Flush the current ordered list
            items.append(line[2:].strip())
            continue
        if line.startswith(". "):  # Check for an ordered list item
            flush_para()  # Flush any previous paragraph
            ordered_items.append(line[2:].strip())  # Add to ordered_items list
            continue
        if not ordered_items and current == []:  # If no ordered items are present and the current block is empty
            flush_ordered_list()  # Flush the ordered list, if any

        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())

    flush_para()
    flush_list()
    # BEGIN: Add ordered list support
    flush_ordered_list()  # Flush any remaining ordered list
    # END: Add ordered list support

    return "\n".join(blocks)
