if fenced_code_block is not None:
            if line.strip() == "```":
                blocks.append("<pre><code>" + "\n".join(fenced_code_block) + "</code></pre>")
                fenced_code_block = None
            else:
                fenced_code_block.append(line)
        elif line.startswith("```"):
            if fenced_code_block is not None:
                continue
            blocks.append("<pre><code>")
            fenced_code_block = []
        else:
            flush_para()
            flush_list()

    # Existing code continues below this point, unchanged.
    if current:
        blocks.append("<p>" + _inline(" ".join(current)) + "</p>")
    return "\n".join(blocks)
