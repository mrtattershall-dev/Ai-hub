# BEGIN: Add ordered list support
            if line.startswith(". "):  # Check for an ordered list item
                flush_para()  # Flush the current paragraph, if any
                ordered_items.append(line[2:].strip())  # Append to ordered_items list
                continue

            if not ordered_items and current == []:  # If no ordered items are present and current block is empty
                flush_ordered_list()  # Flush the ordered list, if any
            # END: Add ordered list support
        elif line.startswith(". "):  # Check for an ordered list item
            flush_para()  # Flush the current paragraph, if any
            ordered_items.append(line[2:].strip())  # Append to ordered_items list
            continue

        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
    flush_para()
    flush_list()
    flush_ordered_list()  # Flush the ordered list, if any
    return "\n".join(blocks)
