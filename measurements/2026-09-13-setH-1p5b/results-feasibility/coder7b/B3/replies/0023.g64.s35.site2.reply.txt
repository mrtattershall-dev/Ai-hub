    def flush_ol():

            flush_list()

        if line.startswith("- "):

            items.append(line[2:].strip())

        elif line.startswith("1. "):  # BEGIN SOLUTION

    flush_list()
    flush_ol()  # Flush ordered lists at the end of processing all lines
