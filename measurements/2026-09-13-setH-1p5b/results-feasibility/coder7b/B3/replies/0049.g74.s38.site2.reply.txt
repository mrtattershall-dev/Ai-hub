# END CODE ADDED FOR FENCED CODE BLOCKS
        if line.startswith("```"):
            flush_para()
            flush_list()
            if fenced_code_block is None:
                fenced_code_block = True
                continue
            else:
                blocks.append("<pre><code>" + "\n".join(current).replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;") + "</code></pre>")
                del current[:]
                fenced_code_block = None
                continue

        if fenced_code_block:
            current.append(line)
            continue

        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())

    flush_para()
    flush_list()

    return "\n".join(blocks)
