fenced_code_block = None
    for line in str(text).split("\n"):
        if line.strip() == "":
            flush_para()
            flush_list()
            continue
        if line.startswith("- "):
            flush_para()
            items.append(line[2:].strip())
            continue
        flush_list()
        h = _is_heading(line)
        if h:
            flush_para()
            level, body = h
            blocks.append("<h%d>%s</h%d>" % (level, _inline(body), level))
            continue
        current.append(line.strip())
        # BEGIN NEW CODE BLOCK FOR FENCED CODE BLOCKS
        if line.startswith("```"):
            if fenced_code_block is None:
                fenced_code_block = []
            else:
                blocks.append("<pre><code>{}</code></pre>".format("\n".join(fenced_code_block)))
                fenced_code_block = None
                continue
        elif fenced_code_block is not None:
            fenced_code_block.append(line)
    if fenced_code_block is not None:
        blocks.append("<pre><code>{}</code></pre>".format("\n".join(fenced_code_block)))
    # END NEW CODE BLOCK FOR FENCED CODE BLOCKS
    return "\n".join(blocks)
