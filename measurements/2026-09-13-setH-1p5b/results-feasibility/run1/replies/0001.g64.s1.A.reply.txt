def to_markdown(s):
    text = s
    # Replace <p> with blank line blocks
    text = re.sub(r"<p>([^<]+)</p>", r"\1\n", text)
    # Replace <h1>, <h2>, and <h3> with 1-3 '#' + space
    text = re.sub(r"<h([1-3])>([^<]+)</h\1>", r"#{1,3} \2", text)
    # Replace <strong> and *em*; unpaired markers stay literal
    text = re.sub(r"<strong>([^<]+)</strong>", r"**\1**", text)
    text = re.sub(r"<em>([^<]+)</em>", r"*\1*", text)
    # Replace <code> with `code` -> <code>, escaped, and * inside it is not emphasis
    text = re.sub(r"<code>([^`]+)`", r"`\1`", text)
    # Replace <a href="url">text</a> -> [text](url), a double quote in the url as &quot;
    text = re.sub(r'<a href="(.*?)">(.*?)</a>', r'[\2](\1)', text)
    return text
