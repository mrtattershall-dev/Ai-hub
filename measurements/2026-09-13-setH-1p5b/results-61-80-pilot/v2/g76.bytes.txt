import heapq
from collections import deque


class Graph:
    def __init__(self):
        self._adj = {}

    def add_node(self, n):
        self._adj.setdefault(n, {})

    def add_edge(self, a, b, weight=1):
        if not isinstance(weight, (int, float)) or isinstance(weight, bool) or weight <= 0:
            raise ValueError("weight must be a positive number")
        self.add_node(a)
        self.add_node(b)
        self._adj[a][b] = weight

    def nodes(self):
        return sorted(self._adj)

    def neighbors(self, a):
        if a not in self._adj:
            raise KeyError(a)
        return sorted(self._adj[a].items())

    def bfs(self, start):
        if start not in self._adj:
            raise KeyError(start)
        seen = {start}
        order = [start]
        q = deque([start])
        while q:
            cur = q.popleft()
            for nxt, _w in sorted(self._adj[cur].items()):
                if nxt not in seen:
                    seen.add(nxt)
                    order.append(nxt)
                    q.append(nxt)
        return order

    def shortest_path(self, a, b):
        if a not in self._adj:
            raise KeyError(a)
        if b not in self._adj:
            raise KeyError(b)
        if a == b:
            return (0, [a])
        dist = {a: 0}
        prev = {}
        heap = [(0, a)]
        done = set()
        while heap:
            d, cur = heapq.heappop(heap)
            if cur in done:
                continue
            done.add(cur)
            if cur == b:
                path = [b]
                while path[-1] != a:
                    path.append(prev[path[-1]])
                path.reverse()
                return (d, path)
            for nxt, w in sorted(self._adj[cur].items()):
                nd = d + w
                if nxt not in dist or nd < dist[nxt]:
                    dist[nxt] = nd
                    prev[nxt] = cur
                    heapq.heappush(heap, (nd, nxt))
        return None

    def has_cycle(self):
        WHITE, GREY, BLACK = 0, 1, 2
        colour = {n: WHITE for n in self._adj}

        def visit(n):
            colour[n] = GREY
            for nxt in self._adj[n]:
                if colour.get(nxt, WHITE) == GREY:
                    return True
                if colour.get(nxt, WHITE) == WHITE and visit(nxt):
                    return True
            colour[n] = BLACK
            return False

        return any(colour[n] == WHITE and visit(n) for n in list(self._adj))

    # goal 46: Kahn's algorithm with a MIN-HEAP, so the smallest ready node is always taken next.
    def topo_order(self):
        indeg = {n: 0 for n in self._adj}
        for a in self._adj:
            for b in self._adj[a]:
                indeg[b] = indeg.get(b, 0) + 1
        ready = [n for n in self._adj if indeg[n] == 0]
        heapq.heapify(ready)
        out = []
        while ready:
            n = heapq.heappop(ready)
            out.append(n)
            for b in sorted(self._adj[n]):
                indeg[b] -= 1
                if indeg[b] == 0:
                    heapq.heappush(ready, b)
        if len(out) != len(self._adj):
            raise ValueError("graph has a cycle")
        return out

    # goal 56: drop the node and every edge pointing at it.
    def remove_node(self, n):
        if n not in self._adj:
            raise KeyError(n)
        del self._adj[n]
        for a in self._adj:
            self._adj[a].pop(n, None)

    def components(self):
        seen = set()
        out = []
        for n in self._adj:
            if n not in seen:
                self.dfs(n, seen, out)
        return out

    def dfs(self, n, seen, out):
        if n not in self._adj:
            raise KeyError(n)
        if n in seen:
            return
        seen.add(n)
        out.append(n)
        for nxt in sorted(self._adj[n].items()):
            self.dfs(nxt[0], seen, out)
