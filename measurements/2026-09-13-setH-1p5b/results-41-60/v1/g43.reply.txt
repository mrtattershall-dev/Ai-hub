class Matrix {
  constructor(rows, cols) {
    this.rows = rows;
    this.cols = cols;
    this.data = new Array(rows).fill(null).map(() => new Array(cols).fill(0));
  }

  set(row, col, value) {
    if (row < 0 || row >= this.rows || col < 0 || col >= this.cols) {
      throw new Error('Row or column index out of bounds');
    }
    this.data[row][col] = value;
  }

  get(row, col) {
    if (row < 0 || row >= this.rows || col < 0 || col >= this.cols) {
      throw new Error('Row or column index out of bounds');
    }
    return this.data[row][col];
  }

  equals(other, eps = 1e-9) {
    if (!other instanceof Matrix) {
      throw new Error('Parameter must be a Matrix instance');
    }
    if (this.rows !== other.rows || this.cols !== other.cols) {
      return false;
    }
    for (let i = 0; i < this.rows; i++) {
      for (let j = 0; j < this.cols; j++) {
        if (Math.abs(this.get(i, j) - other.get(i, j)) > eps) {
          return false;
        }
      }
    }
    return true;
  }
