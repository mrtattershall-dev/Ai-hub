```python
import numpy as np

class LogAnalyzer:
    def __init__(self, log_entries):
        self.log_entries = log_entries

    def percentile(self, p):
        if not (0 < p <= 100):
            raise ValueError("p must be between 0 and 100")
        if len(self.log_entries) == 0:
            raise ValueError("No entries to calculate the percentile for")

        sorted_logs = np.sort(self.log_entries)
        rank = int(p / 100 * (len(sorted_logs) - 1)) + 1
        return sorted_logs[rank - 1]

# Example usage
if __name__ == "__main__":
    log_entries = [1, 3, 5, 7, 9]
    analyzer = LogAnalyzer(log_entries)
    p = 50  # Calculate the 50th percentile
    result = analyzer.percentile(p)
    print(f"The {p}th percentile of the seconds values is: {result}")
```
