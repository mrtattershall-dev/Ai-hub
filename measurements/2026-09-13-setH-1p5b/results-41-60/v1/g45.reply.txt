```javascript
// s5_expr.js

function min(...args) {
  if (args.length === 0) throw new Error('min() requires at least one argument');
  return Math.min(...args);
}

function max(...args) {
  if (args.length === 0) throw new Error('max() requires at least one argument');
  return Math.max(...args);
}

function abs(x) {
  if (x < 0) throw new Error(`abs(${x})`);
  return Math.abs(x);
}

function sqrt(x) {
  if (x < 0) throw new Error(`sqrt(${x})`);
  return Math.sqrt(x);
}

module.exports = { min, max, abs, sqrt };
```
