) {
    if (this.rows.length !== this.rows[0].length) throw new Error('not a square matrix');
    const n = this.rows.length;
    let det = 1;
    for (let i = 0; i < n; i++) {
      const x = this.rows[i][i];
      if (!Number.isFinite(x)) throw new Error('non-finite number');
      if (x === 0) return 0;
      det *= x;
      for (let j = i + 1; j < n; j++) {
        const y = this.rows[j][i];
        if (!Number.isFinite(y)) throw new Error('non-finite number');
        const subdet = this._combine(this.rows.slice(0, j), (a, b) => a - b).determinant();
        det *= -y * subdet;
      }
    }
    return det;
  }
