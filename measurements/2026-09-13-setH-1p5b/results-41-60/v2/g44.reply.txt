# s4_markdown.py

def convert_to_html(text):
    # Replace [text](url) with <a href="url">text</a>
    text = text.replace('[', '<').replace(']', '>')
    text = text.replace('{', '&{').replace('}', '}')
    return text

# Example usage
input_text = "[Hello](https://www.example.com)"
output_html = convert_to_html(input_text)
print(output_html)  # Output: <a href="https://www.example.com">Hello</a>
