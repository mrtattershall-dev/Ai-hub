#   goal 28  module-level letter(percent) and Gradebook.report()
#   goal 38  add_assignment(..., category="default"), set_weight(category, weight);
#            percent becomes the weighted average across the categories the student has scores in,
#            with weights rescaled over those categories. A single category must not change it.


def letter(percent):
    if percent >= 90:
        return "A"
    if percent >= 80:
        return "B"
    if percent >= 70:
        return "C"
    if percent >= 60:
        return "D"
    return "F"


class Gradebook:
    def __init__(self):
        self._students = []
        self._assignments = {}          # name -> {"max": float, "category": str}
        self._scores = {}               # (student, assignment) -> points
        self._weights = {}              # category -> weight

    def add_student(self, name):
        if name in self._students:
            raise ValueError("student already exists")
        self._students.append(name)

    def add_assignment(self, name, max_points, category="default"):
        if name in self._assignments:
            raise ValueError("assignment already exists")
        if not isinstance(max_points, (int, float)) or isinstance(max_points, bool) or max_points <= 0:
            raise ValueError("max_points must be a positive number")
        self._assignments[name] = {"max": max_points, "category": category}

    def set_weight(self, category, weight):
        if not isinstance(weight, (int, float)) or isinstance(weight, bool) or weight <= 0:
            raise ValueError("weight must be a positive number")
        self._weights[category] = weight

    def record(self, student, assignment, points):
        if student not in self._students:
            raise KeyError(student)
        if assignment not in self._assignments:
            raise KeyError(assignment)
        top = self._assignments[assignment]["max"]
        if not isinstance(points, (int, float)) or isinstance(points, bool) or points < 0 or points > top:
            raise ValueError("points must be between 0 and max_points")
        self._scores[(student, assignment)] = points

    def score(self, student, assignment):
        return self._scores.get((student, assignment))

    def percent(self, student):
        if student not in self._students:
            raise KeyError(student)
        # Gather per-category totals over the assignments this student actually has scores for.
        totals = {}
        for (s, a), pts in self._scores.items():
            if s != student:
                continue
            cat = self._assignments[a]["category"]
            got, top = totals.get(cat, (0.0, 0.0))
            totals[cat] = (got + pts, top + self._assignments[a]["max"])
        if not totals:
            return None
        weighted = 0.0
        total_weight = 0.0
        for cat, (got, top) in totals.items():
            w = self._weights.get(cat, 1)
            weighted += w * (got / top * 100.0)
            total_weight += w
        return round(weighted / total_weight, 2)

    def report(self):
        out = []
        for s in sorted(self._students):
            p = self.percent(s)
            out.append((s, p, "-" if p is None else letter(p)))
        return out

    def drop_lowest(self):
        scores = self._scores.copy()         # shallow copy for in-place modification
        keys = list(scores.keys())            # to preserve order of assignments
        while keys:
            best = (keys[0], None)
            for k, v in scores.items():
                if k[1] != best[1]:
                    continue
                pts = self._assignments[k[1]]["max"]
                if v > best[0][1] * pts / top or (v == best[0][1] * pts / top and k < best[0]):
                    best = (k, v)
            del scores[keys.pop(0)]
        return scores
