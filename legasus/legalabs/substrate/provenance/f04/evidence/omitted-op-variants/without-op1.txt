# Temperature helpers.


def to_f(c):
    return c * 9.0 / 5.0 + 32.0


def to_c(f):
    return (f - 32.0) * 5.0 / 9.0


def record_c(c):
    _READINGS.append(c)
    return c


def floor_c(c):
    return max(-273.15, c)


def extremes():
    if not _READINGS:
        return None
    return (min(_READINGS), max(_READINGS))
