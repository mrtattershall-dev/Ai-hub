# Small text helpers.


def normalize(s):
    return " ".join(str(s).split()).lower()


def word_count(s):
    return len(normalize(s).split())


def initials(s):
    return "".join(w[0] for w in normalize(s).split() if w)


def counted(s):
    key = normalize(s)
    _SEEN[key] = _SEEN.get(key, 0) + 1
    return _SEEN[key]
